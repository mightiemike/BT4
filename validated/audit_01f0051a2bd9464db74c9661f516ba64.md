#[derive(Debug, Serialize, Deserialize)]
pub struct AggregatorResource<T> {
    value: T,
    max_value: T,
}

impl<T> AggregatorResource<T> {
    pub fn new(value: T, max_value: T) -> Self {
        Self { value, max_value }
    }

    pub fn get(&self) -> &T {
        &self.value
    }

    pub fn set(&mut self, value: T) {
        self.value = value;
    }
}
