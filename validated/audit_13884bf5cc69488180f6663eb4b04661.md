This confirms a real design flaw. The gasless allowlist is purely message-type driven, with no verification that the sender is actually a bonded universal validator, so any unprivileged key can get "free" inclusion for these message types even though the underlying business logic will always reject them.

### Title
Unprivileged accounts can broadcast fee-free `MsgVoteTssKeyProcess`/`MsgVoteFundMigration` (and other gasless-allowlisted messages) that are guaranteed to fail, enabling free spam - (`app/txpolicy/gasless.go`)

### Summary
`txpolicy.IsGaslessTx` grants fee exemption based solely on the message's type URL, not on the identity, bond status, or authorization of the sender.

### Finding Description
`IsGaslessTx` inspects `tx.GetMsgs()` and returns `true` if every message type (including inner `authz.MsgExec` messages) matches a fixed allowlist that includes `MsgVoteTssKeyProcess` and `MsgVoteFundMigration` [1](#0-0) . This function is consumed identically by `DeductFeeDecorator.AnteHandle`, which skips fee deduction entirely for such txs [2](#0-1) , by `MinGasPriceDecorator.AnteHandle`, which skips the minimum-gas-price check [3](#0-2) , and by `AccountInitDecorator`, which will even auto-create a brand-new on-chain account for the sender for free as long as the tx's own signature verifies against the freshly generated key [4](#0-3) .

None of these three decorators check whether the signer is a bonded universal validator. That check only happens later, inside the message handlers: `msgServer.VoteTssKeyProcess` and `msgServer.VoteFundMigration` both call `ms.k.uvalidatorKeeper.IsBondedUniversalValidator(ctx, msg.Signer)` and reject with an error if the signer isn't bonded [5](#0-4) [6](#0-5) .

Consequently, any freshly generated, non-validator key can:
1. Sign a `MsgVoteTssKeyProcess` (or `MsgVoteFundMigration`) with itself as signer.
2. Pass the full ante chain fee-free (`DeductFeeDecorator`, `MinGasPriceDecorator` both short-circuit via `IsGaslessTx`), and even get a new account auto-created on first use.
3. Only fail once execution reaches `msgServer`, after consuming block gas/space and mempool bandwidth.

### Impact Explanation
This allows unlimited, cost-free transaction submission that is guaranteed to fail at the message-execution layer. It is a non-network-level DoS vector: attackers can flood blocks with zero-fee transactions that consume gas limit/block space and mempool resources without ever paying for it, since neither `DeductFeeDecorator` nor `MinGasPriceDecorator` enforces any cost on this message type regardless of sender. This matches the impact category of "denial of service only when it is not network-level and is reachable without privileged control."

### Likelihood Explanation
High — the exploit requires no privileges: any address can generate a new key, construct a syntactically valid `MsgVoteTssKeyProcess`/`MsgVoteFundMigration`, sign it, and broadcast repeatedly with incrementing sequence numbers, all without funding an account (thanks to `AccountInitDecorator`'s free account creation path for gasless txs).

### Recommendation
`IsGaslessTx` (or the decorators consuming it) should additionally verify sender eligibility before granting fee exemption for validator-only gasless message types — e.g., check `IsBondedUniversalValidator` (or an equivalent authorization check) for `MsgVoteTssKeyProcess`/`MsgVoteFundMigration`/`MsgVoteInbound`/`MsgVoteOutbound`/`MsgVoteChainMeta` senders inside the ante handler, rejecting (or requiring normal fees for) non-eligible senders before allowing gasless admission.

### Proof of Concept
1. Generate a new keypair/address `attacker` with zero balance.
2. Construct `MsgVoteTssKeyProcess{Signer: attacker, ...}` (arbitrary key process/process ids), wrap in a `Tx` signed by `attacker`, with `Fee = 0`.
3. Submit via `CheckTx`/`broadcast_tx_sync` through the full `NewCosmosAnteHandler` chain [7](#0-6) .
4. Observe: `AccountInitDecorator` creates the account for free, `DeductFeeDecorator`/`MinGasPriceDecorator` both skip fee/min-gas enforcement via `IsGaslessTx`, tx is accepted into mempool/block with zero fee.
5. `msgServer.VoteTssKeyProcess` rejects with "universal validator ... is not bonded", but only after the tx already consumed block gas/space for free.
6. Repeat with incrementing sequence numbers to demonstrate unbounded free submission throughput.

// IsGaslessTx checks if a transaction contains only allowed gasless message types
// Returns true if all messages in the transaction are in the allowed gasless message types
func IsGaslessTx(tx sdk.Tx) bool {
	var (
		// GaslessMsgTypes defines the message types that are allowed in gasless transactions
		GaslessMsgTypes = []string{
			sdk.MsgTypeURL(&uexecutortypes.MsgMigrateUEA{}),
			sdk.MsgTypeURL(&uexecutortypes.MsgExecutePayload{}),
			sdk.MsgTypeURL(&uexecutortypes.MsgVoteInbound{}),
			sdk.MsgTypeURL(&uexecutortypes.MsgVoteOutbound{}),
			sdk.MsgTypeURL(&utsstypes.MsgVoteTssKeyProcess{}),
			sdk.MsgTypeURL(&utsstypes.MsgVoteFundMigration{}),
			sdk.MsgTypeURL(&uexecutortypes.MsgVoteChainMeta{}),
		}
	)

	msgs := tx.GetMsgs()
	if len(msgs) == 0 {
		return false
	}

	for _, msg := range msgs {
		switch m := msg.(type) {
		case *authz.MsgExec:
			// Only gasless if ALL inner messages are allowed
			for _, innerMsg := range m.Msgs {
				if !slices.Contains(GaslessMsgTypes, innerMsg.TypeUrl) {
					return false
				}
			}
		default:
			if !slices.Contains(GaslessMsgTypes, sdk.MsgTypeURL(msg)) {
				return false
			}
		}
	}
	return true
}

		ctx.Logger().Debug("account init decorator: non-gasless tx, skipping account init")
		return next(ctx, tx, simulate)

	if !ok {
		return ctx, errorsmod.Wrap(sdkerrors.ErrTxDecode, "invalid transaction type")
	}

	if err != nil || len(signers) != 1 {
		ctx.Logger().Debug("account init decorator: could not get unique signer, passing to next handler",
			"num_signers", len(signers),
			"error", err,
		)
		return next(ctx, tx, simulate)
	}

	newAccAddr := signers[0]

				"error", err,

			return ctx, err
		}

		acc.SetSequence(1)
		aid.ak.SetAccount(ctx, acc)
		ctx.Logger().Info("account init decorator: new account created via gasless tx",

// VoteTssKeyProcess implements types.MsgServer.
func (ms msgServer) VoteTssKeyProcess(ctx context.Context, msg *types.MsgVoteTssKeyProcess) (*types.MsgVoteTssKeyProcessResponse, error) {
	signerAccAddr, err := sdk.AccAddressFromBech32(msg.Signer)
	if err != nil {
		return nil, fmt.Errorf("invalid signer address: %w", err)
	}

	// Convert account to validator operator address
	signerValAddr := sdk.ValAddress(signerAccAddr)

	// Lookup the linked universal validator for this signer
	isBonded, err := ms.k.uvalidatorKeeper.IsBondedUniversalValidator(ctx, msg.Signer)
	if err != nil {
		return nil, errors.Wrapf(err, "failed to check bonded status for signer %s", msg.Signer)
	}
	if !isBonded {
		return nil, fmt.Errorf("universal validator for signer %s is not bonded", msg.Signer)
	}

	isTombstoned, err := ms.k.uvalidatorKeeper.IsTombstonedUniversalValidator(ctx, msg.Signer)
	if err != nil {
		return nil, errors.Wrapf(err, "failed to check tombstoned status for signer %s", msg.Signer)
	}
	if isTombstoned {
		return nil, fmt.Errorf("universal validator for signer %s is tombstoned", msg.Signer)
	}

	err = ms.k.VoteTssKeyProcess(ctx, signerValAddr, msg.TssPubkey, msg.KeyId, msg.ProcessId)
	if err != nil {
		return nil, err
	}

	return &types.MsgVoteTssKeyProcessResponse{}, nil
}
