Protobuf-go's decoding paths operate on immutable, fully-buffered `[]byte` slices (e.g., `unmarshalPointerEager` in [1](#0-0) , `ConsumeFieldValue` in [2](#0-1) ) or use `io.ReadFull`/`bufio.Peek` with an explicit, upfront-determined size before parsing, as in `protodelim.UnmarshalFrom` [3](#0-2) . None of these loops depend on re-reading a mutable, attacker-shrinkable resource (like a file being concurrently modified) whose changing size could desynchronize a length counter from actual remaining bytes and cause an infinite loop, which is the core broken invariant in the GNU Tar CVE. All buffer/index-based loops (`BufferReader`, `unmarshalCounts`, `unmarshalSeed`, lazy extension/message decoding) advance strictly through a fixed-length in-memory slice with monotonically decreasing length checks (`len(b) > 0`), so there's no reachable unprivileged-request-path analog to the tar sparse-dump infinite-loop bug in this codebase.

func consumeFieldValueD(num Number, typ Type, b []byte, depth int) (n int) {
	switch typ {
	case VarintType:
		_, n = ConsumeVarint(b)
		return n
	case Fixed32Type:
		_, n = ConsumeFixed32(b)
		return n
	case Fixed64Type:
		_, n = ConsumeFixed64(b)
		return n
	case BytesType:
		_, n = ConsumeBytes(b)
		return n
	case StartGroupType:
		if depth < 0 {
			return errCodeRecursionDepth
		}
		n0 := len(b)
		for {
			num2, typ2, n := ConsumeTag(b)
			if n < 0 {
				return n // forward error code
			}
			b = b[n:]
			if typ2 == EndGroupType {
				if num != num2 {
					return errCodeEndGroup
				}
				return n0 - len(b)
			}

			n = consumeFieldValueD(num2, typ2, b, depth-1)
			if n < 0 {
				return n // forward error code
			}
			b = b[n:]
		}
	case EndGroupType:
		return errCodeEndGroup
	default:
		return errCodeReserved
	}
}
