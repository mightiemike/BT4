No vulnerability found.

Based on the investigation, the reported bug class—writing past a heap-allocated output buffer during decompression due to an unchecked length field (`ndlz8_decompress` in c-blosc2)—does not have an analogous unchecked-write path in protobuf-go's decode paths.

Every location that consumes untrusted length-prefixed data in the scoped decode paths performs an explicit bounds check before slicing or copying:

- `protowire.ConsumeBytes` validates `m > uint64(len(b[n:]))` before returning the sliced sub-buffer, rejecting truncated/oversized lengths. [1](#0-0) 
- All generated `consume*Slice` coders (e.g. `consumeUint64Slice`, `consumeSint64Slice`, `consumeFixed32Slice`) call `protowire.ConsumeBytes`/`ConsumeVarint`/`ConsumeFixed32` and check `n < 0` before appending to the destination slice, so no over-read/over-write of the destination slice occurs. [2](#0-1) 
- `protolazy.BufferReader.Skip` explicitly checks `len(b.Buf) < b.Pos+n` before advancing, and `SkipBytes`/`SkipValue`/`SkipGroup` route through it, preventing out-of-bounds skips during lazy-decode indexing. [3](#0-2) 
- `protodelim.UnmarshalFrom`, which reads a varint size prefix and then allocates/reads exactly that many bytes, enforces a `MaxSize` (default 4 MiB) check on the declared size before allocation, and validates `size > math.MaxInt` when the limit is disabled, preventing unbounded/overflowing allocation. [4](#0-3) 
- The eager/lazy message unmarshal dispatch (`unmarshalPointer`, `unmarshalPointerLazy`, `validate`) enforces a recursion-depth counter on every message/group/map descent, and the lazy-field tag/length parsing likewise checks `n < 0` before advancing. [5](#0-4) [6](#0-5) 

The report describes a codec (`ndlz8_decompress`) writing output based on an internal decompressed-length calculation without validating it against the allocated output buffer size — a pattern specific to c-blosc2's custom LZ-style codec. protobuf-go has no equivalent "decompression into a fixed-size output buffer" step; all wire-format consumption is bounds-checked against the actual remaining input slice length before any slice/copy operation, and Go's slice/append semantics prevent writing past allocated capacity without a panic (not a silent heap overflow). No matching unchecked-write sink was found on any reachable default-parser (binary or ProtoJSON) path.

// ConsumeBytes parses b as a length-prefixed bytes value, reporting its length.
// This returns a negative length upon an error (see [ParseError]).
func ConsumeBytes(b []byte) (v []byte, n int) {
	m, n := ConsumeVarint(b)
	if n < 0 {
		return nil, n // forward error code
	}
	if m > uint64(len(b[n:])) {
		return nil, errCodeTruncated
	}
	return b[n:][:m], n + int(m)
}

func consumeUint64Slice(b []byte, p pointer, wtyp protowire.Type, f *coderFieldInfo, opts unmarshalOptions) (out unmarshalOutput, err error) {
	sp := p.Uint64Slice()
	if wtyp == protowire.BytesType {
		b, n := protowire.ConsumeBytes(b)
		if n < 0 {
			return out, errDecode
		}
		count := 0
		for _, v := range b {
			if v < 0x80 {
				count++
			}
		}
		if count > 0 {
			p.growUint64Slice(count)
		}
		s := *sp
		for len(b) > 0 {
			var v uint64
			var n int
			if len(b) >= 1 && b[0] < 0x80 {
				v = uint64(b[0])
				n = 1
			} else if len(b) >= 2 && b[1] < 128 {
				v = uint64(b[0]&0x7f) + uint64(b[1])<<7
				n = 2
			} else {
				v, n = protowire.ConsumeVarint(b)
			}
			if n < 0 {
				return out, errDecode
			}
			s = append(s, v)
			b = b[n:]
		}
		*sp = s
		out.n = n
		return out, nil

// skip skips the specified number of bytes
func (b *BufferReader) Skip(n int) (err error) {
	if len(b.Buf) < b.Pos+n {
		return io.ErrUnexpectedEOF
	}
	b.Pos += n
	return
}

// skipFixed64 skips a fixed64
func (b *BufferReader) SkipFixed64() (err error) {
	return b.Skip(8)
}

// skipFixed32 skips a fixed32
func (b *BufferReader) SkipFixed32() (err error) {
	return b.Skip(4)
}

// skipBytes skips a set of bytes
func (b *BufferReader) SkipBytes() (err error) {
	n, err := b.DecodeVarint32()
	if err != nil {
		return err
	}
	return b.Skip(int(n))
}
