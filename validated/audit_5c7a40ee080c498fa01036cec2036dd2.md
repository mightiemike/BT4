The CVE describes a use-after-free in OpenSSH caused by concurrent remote-forwarding operations racing on a `realloc`'d buffer — a concurrency bug in mutable shared state accessed by two operations from the same peer connection.

Searching protobuf-go for an analogous broken invariant (buffer reuse/realloc racing with concurrent access) turns up only the lazy-unmarshal buffer-sharing machinery in `internal/impl/lazy.go` and `internal/protolazy/lazy.go`, plus the dedicated race tests in `internal/race_test/lazy/lazy_race_test.go` [1](#0-0) [2](#0-1) . This code is explicitly designed and tested to allow safe *concurrent reads* of a lazily-decoded message (via atomic pointer/bitmap operations) [3](#0-2) , but protobuf-go's documented threat model — like essentially all Go proto libraries — never guarantees safety for concurrent *writes*/mutations of the same message; that is explicitly the caller's responsibility, not a memory-safety bug in the library. There is no reachable single unprivileged request path in protobuf-go where two decode/marshal operations on the *same* message object are triggered internally by the library itself (unlike the SSH case where the forwarding subsystem itself triggers concurrent realloc operations on shared server state). The `MarshalAppend`/`growcap` buffer growth path in `proto/encode.go` also does not exhibit a UAF, since realloc-and-copy semantics of Go's `append`/`make`+`copy` don't leave a dangling pointer to freed memory the way C's `realloc` can [4](#0-3) .

Since the analog requires either (a) attacker-controlled concurrent mutation of a shared message from a single request path, which protobuf-go's threat model explicitly excludes as "trusted in-memory misuse," or (b) a genuine internal realloc/UAF condition, which doesn't exist in Go's garbage-collected slice semantics, this report does not map to an exploitable protobuf-go analog.

func (mi *MessageInfo) lazyUnmarshal(p pointer, num protoreflect.FieldNumber) {
	var f *coderFieldInfo
	if int(num) < len(mi.denseCoderFields) {
		f = mi.denseCoderFields[num]
	} else {
		f = mi.coderFields[num]
	}
	if f == nil {
		panic(fmt.Sprintf("lazyUnmarshal: field info for %v.%v", mi.Desc.FullName(), num))
	}
	lazy := *p.Apply(mi.lazyOffset).LazyInfoPtr()
	start, end, found, _, multipleEntries := lazy.FindFieldInProto(uint32(num))
	if !found && multipleEntries == nil {
		panic(fmt.Sprintf("lazyUnmarshal: can't find field data for %v.%v", mi.Desc.FullName(), num))
	}
	// The actual pointer in the message can not be set until the whole struct is filled in, otherwise we will have races.
	// Create another pointer and set it atomically, if we won the race and the pointer in the original message is still nil.
	fp := pointerOfValue(reflect.New(f.ft))
	if multipleEntries != nil {
		for _, entry := range multipleEntries {
			mi.unmarshalField(lazy.Buffer()[entry.Start:entry.End], fp, f, lazy, lazy.UnmarshalFlags())
		}
	} else {
		mi.unmarshalField(lazy.Buffer()[start:end], fp, f, lazy, lazy.UnmarshalFlags())
	}
	p.Apply(f.offset).AtomicSetPointerIfNil(fp.Elem())
}

// growcap scales up the capacity of a slice.
//
// Given a slice with a current capacity of oldcap and a desired
// capacity of wantcap, growcap returns a new capacity >= wantcap.
//
// The algorithm is mostly identical to the one used by append as of Go 1.14.
func growcap(oldcap, wantcap int) (newcap int) {
	if wantcap > oldcap*2 {
		newcap = wantcap
	} else if oldcap < 1024 {
		// The Go 1.14 runtime takes this case when len(s) < 1024,
		// not when cap(s) < 1024. The difference doesn't seem
		// significant here.
		newcap = oldcap * 2
	} else {
		newcap = oldcap
		for 0 < newcap && newcap < wantcap {
			newcap += newcap / 4
		}
		if newcap <= 0 {
			newcap = wantcap
		}
	}
	return newcap
}
