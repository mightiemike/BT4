fn floored_triangular_root(a: usize) -> usize {
    // Solve `n*(n+1)/2 <= a`, or equivalently `n^2 + n - 2a <= 0`
    let discriminant = 1 + 8 * a;
    let sqrt_disc = discriminant.sqrt(); // integer sqrt
    (sqrt_disc - 1) / 2
}
