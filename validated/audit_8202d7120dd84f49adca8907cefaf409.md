The fee-on-transfer bug class does not map onto this codebase. Citrea's bridge value is `cBTC`, the chain's native asset (not an ERC-20 balance read via `balanceOf`), and the deposit/withdrawal amount is a fixed constant `depositAmount` set at `initialize` time rather than derived from a before/after balance diff on a transferred token [1](#0-0) . Native `.call{value: depositAmount}("")` transfers in the EVM cannot be skimmed by a "fee-on-transfer" mechanism the way an ERC-20 `transferFrom` can, and the only ERC-20-like token in scope, `WCBTC9`, is a standard 1:1 wrapper with no fee logic [2](#0-1) . There is no equality binding (minted cBTC vs. actual BTC deposited) that depends on reading a mutable ERC-20 balance before/after a transfer, so the reported bug class has no reachable analog that breaks a rollup binding here.

    function deposit() public payable {
        balanceOf[msg.sender] += msg.value;
        emit Deposit(msg.sender, msg.value);
    }
    function withdraw(uint wad) public {
        require(balanceOf[msg.sender] >= wad);
        balanceOf[msg.sender] -= wad;
        msg.sender.transfer(wad);
        emit Withdrawal(msg.sender, wad);
    }

    function totalSupply() public view returns (uint) {
        return address(this).balance;
    }

    function approve(address guy, uint wad) public returns (bool) {
        allowance[msg.sender][guy] = wad;
        emit Approval(msg.sender, guy, wad);
        return true;
    }

    function transfer(address dst, uint wad) public returns (bool) {
        return transferFrom(msg.sender, dst, wad);
    }

    function transferFrom(address src, address dst, uint wad)
        public
        returns (bool)
    {
        require(balanceOf[src] >= wad);

        if (src != msg.sender && allowance[src][msg.sender] != uint(-1)) {
            require(allowance[src][msg.sender] >= wad);
            allowance[src][msg.sender] -= wad;
        }

        balanceOf[src] -= wad;
        balanceOf[dst] += wad;

        emit Transfer(src, dst, wad);

        return true;
