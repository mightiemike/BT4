                let field_ty_count = NumTypeNodes::new(field_ty.num_nodes() as u64);
                ((field_ty, field_ty_count), (struct_type, struct_ty_count))
            });
        Ok(((field_ty, *field_ty_count), (struct_ty, *struct_ty_count)))
    }
