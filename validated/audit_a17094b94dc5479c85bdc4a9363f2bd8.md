The path-traversal bug class in the report (constructing a filesystem path from unsanitized user input and passing it to a file-open sink) does not have a reachable analog in protobuf-go's production, unprivileged request path. The file-path-like operations found in this codebase fall into categories excluded by the rules:

- `internal/cmd/generate-protos/main.go`, `internal/cmd/generate-types/main.go`, and `internal/cmd/pbdump/pbdump.go` are developer/build tooling, not reachable production request paths, and take trusted local flags/CLI args as input [1](#0-0) .
- `reflect/protoregistry.Files.FindFileByPath` and `protodesc.FileOptions.New` resolve import paths from a `FileDescriptorProto`, but per the rules this requires a custom resolver or attacker-supplied descriptor, which is out of scope for the trusted-schema, default-parser threat model [2](#0-1) [3](#0-2) .
- `internal/impl/legacy_file.go`'s `legacyLoadFileDesc` decompresses and unmarshals a legacy gzip-encoded descriptor, but it does not construct or open any filesystem path from message content — it only feeds bytes into `filedesc.Builder` [4](#0-3) .

None of these match the report's pattern of unsanitized user-controlled input flowing into a filesystem path used to read/write files on a standard binary/ProtoJSON decode path.

func (r *Files) FindFileByPath(path string) (protoreflect.FileDescriptor, error) {
	if r == nil {
		return nil, NotFound
	}
	if r == GlobalFiles {
		globalMutex.RLock()
		defer globalMutex.RUnlock()
	}
	fds := r.filesByPath[path]
	switch len(fds) {
	case 0:
		return nil, NotFound
	case 1:
		return fds[0], nil
	default:
		return nil, errors.New("multiple files named %q", path)
	}
}

func legacyLoadFileDesc(b []byte) protoreflect.FileDescriptor {
	// Fast-path: check whether we already have a cached file descriptor.
	if fd, ok := legacyFileDescCache.Load(&b[0]); ok {
		return fd.(protoreflect.FileDescriptor)
	}

	// Slow-path: decompress and unmarshal the file descriptor proto.
	zr, err := gzip.NewReader(bytes.NewReader(b))
	if err != nil {
		panic(err)
	}
	b2, err := io.ReadAll(zr)
	if err != nil {
		panic(err)
	}

	fd := filedesc.Builder{
		RawDescriptor: b2,
		FileRegistry:  resolverOnly{protoregistry.GlobalFiles}, // do not register back to global registry
	}.Build().File
	if fd, ok := legacyFileDescCache.LoadOrStore(&b[0], fd); ok {
		return fd.(protoreflect.FileDescriptor)
	}
	return fd
}
