func (k QuerierV2) GetUniversalTx(goCtx context.Context, req *typesv2.QueryGetUniversalTxRequest) (*typesv2.QueryGetUniversalTxResponse, error) {
	if req == nil {
		return nil, status.Error(codes.InvalidArgument, "invalid request")
	}

	ctx := sdk.UnwrapSDKContext(goCtx)

	currentTx, err := k.UniversalTx.Get(ctx, strings.TrimPrefix(req.Id, "0x"))
	if err != nil {
		if errors.Is(err, collections.ErrNotFound) {
			return nil, status.Error(codes.NotFound, "UniversalTx not found")
		}
		return nil, err
	}

	return &typesv2.QueryGetUniversalTxResponse{
		UniversalTx: &currentTx,
	}, nil
}

func (k Keeper) CreateUniversalTx(ctx context.Context, key string, utx types.UniversalTx) error {
	exists, err := k.UniversalTx.Has(ctx, key)
	if err != nil {
		return err
	}
	if exists {
		return fmt.Errorf("universal tx with key %s already exists", key)
	}

	k.Logger().Debug("universal tx stored", "utx_key", key)
	return k.UniversalTx.Set(ctx, key, utx)
}
