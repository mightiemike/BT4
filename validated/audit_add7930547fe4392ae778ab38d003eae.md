The self-merge duplication behavior in `proto.Merge` is explicit, intended, and documented: merging a message into itself doubles repeated/list and unknown fields, since `Merge` semantics for lists is "append" rather than "replace." This is exercised directly by `TestMergeSelf` in [1](#0-0) , which asserts the doubling as correct, expected output — not a bug. The doc comment on `Merge` itself states that "elements of every list field in src is appended to the corresponding list fields in dst," making this documented behavior rather than an unintended accounting-invariant violation like the Cauldron `stir` bug [2](#0-1) .

Map fields specifically use replace-not-append semantics (`dst.Set(k, ...)`), so self-merging a map field is idempotent and does not duplicate map entries [3](#0-2) . The internal fast-path map merge functions (`mergeMap`, `mergeMapOfBytes`, `mergeMapOfMessage`) also use `SetMapIndex`, which overwrites rather than duplicates entries even in a self-merge scenario [4](#0-3) .

There is no reachable unprivileged request path (default binary/ProtoJSON decode) where this self-merge duplication would corrupt a security-relevant accounting invariant analogous to the Cauldron `ink`/`art` balance duplication — the analog here is a well-tested, intentional API semantic (merge-into-self doubles repeated data), not a caching bug on a ledger. Per the rules, this must be rejected as it does not represent a genuine security flaw but expected, documented, and tested library behavior.

// Merge merges src into dst, which must be a message with the same descriptor.
//
// Populated scalar fields in src are copied to dst, while populated
// singular messages in src are merged into dst by recursively calling Merge.
// The elements of every list field in src is appended to the corresponded
// list fields in dst. The entries of every map field in src is copied into
// the corresponding map field in dst, possibly replacing existing entries.
// The unknown fields of src are appended to the unknown fields of dst.
//
// It is semantically equivalent to unmarshaling the encoded form of src
// into dst with the [UnmarshalOptions.Merge] option specified.

func (o mergeOptions) mergeMap(dst, src protoreflect.Map, fd protoreflect.FieldDescriptor) {
	// Merge semantics replaces, rather than merges into existing entries.
	src.Range(func(k protoreflect.MapKey, v protoreflect.Value) bool {
		switch {
		case fd.Message() != nil:
			dstv := dst.NewValue()
			o.mergeMessage(dstv.Message(), v.Message())
			dst.Set(k, dstv)
		case fd.Kind() == protoreflect.BytesKind:
			dst.Set(k, o.cloneBytes(v))
		default:
			dst.Set(k, v)
		}
		return true
	})
}

func mergeMap(dst, src pointer, f *coderFieldInfo, opts mergeOptions) {
	dstm := dst.AsValueOf(f.ft).Elem()
	srcm := src.AsValueOf(f.ft).Elem()
	if srcm.Len() == 0 {
		return
	}
	if dstm.IsNil() {
		dstm.Set(reflect.MakeMap(f.ft))
	}
	iter := srcm.MapRange()
	for iter.Next() {
		dstm.SetMapIndex(iter.Key(), iter.Value())
	}
}

func mergeMapOfMessage(dst, src pointer, f *coderFieldInfo, opts mergeOptions) {
	dstm := dst.AsValueOf(f.ft).Elem()
	srcm := src.AsValueOf(f.ft).Elem()
	if srcm.Len() == 0 {
		return
	}
	if dstm.IsNil() {
		dstm.Set(reflect.MakeMap(f.ft))
	}
	iter := srcm.MapRange()
	for iter.Next() {
		val := reflect.New(f.ft.Elem().Elem())
		if f.mi != nil {
			f.mi.mergePointer(pointerOfValue(val), pointerOfValue(iter.Value()), opts)
		} else {
			opts.Merge(asMessage(val), asMessage(iter.Value()))
		}
		dstm.SetMapIndex(iter.Key(), val)
	}
}
