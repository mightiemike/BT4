Rationale: The CVE describes a C/C++ heap buffer overflow in Thunderbird's iCal string-dequoting routine (`icalmemory_strdup_and_dequote`), a memory-safety bug class specific to manual buffer/length arithmetic in unmanaged memory. protobuf-go's analogous string-unescaping code paths — `internal/encoding/text/decode_string.go`'s `parseString`/`parseStringValue` [1](#0-0)  and `internal/encoding/json/decode_string.go`'s `parseString` [2](#0-1)  — rely on Go's bounds-checked slice/append operations rather than manual pointer/length buffer management, so the same "unbounded write past allocated buffer" bug class cannot occur; any out-of-bounds access would be caught by the Go runtime and result in a panic (a bounded availability effect, not a heap overflow/memory-corruption issue), which the analog-scan rules explicitly instruct rejecting as a generic panic rather than accepting as equivalent to a C buffer overflow. No sink comparable to `icalmemory_strdup_and_dequote`'s unsafe `strdup`-based buffer sizing exists on a reachable, unprivileged, trusted-schema protobuf-go decode path (binary `protowire.ConsumeBytes` [3](#0-2)  and ProtoJSON/ProtoText string decoders all copy into freshly-sized Go slices/strings with growth handled safely by `append`).

// consumeStringPtr wire decodes a *string pointer as a String.
func consumeStringPtr(b []byte, p pointer, wtyp protowire.Type, f *coderFieldInfo, opts unmarshalOptions) (out unmarshalOutput, err error) {
	if wtyp != protowire.BytesType {
		return out, errUnknown
	}
	v, n := protowire.ConsumeBytes(b)
	if n < 0 {
		return out, errDecode
	}
	vp := p.StringPtr()
	if *vp == nil {
		*vp = new(string)
	}
	**vp = string(v)
	out.n = n
	return out, nil
