    pub fn add_resource_op(&mut self, struct_tag: StructTag, op: Op<Resource>) -> Result<()> {
        use btree_map::Entry::*;

        match self.resources.entry(struct_tag) {
            Occupied(entry) => bail!(
                "Resource {} already exists",
                entry.key().to_canonical_string()
            ),
            Vacant(entry) => {
                entry.insert(op);
            },
        }

        Ok(())
    }
