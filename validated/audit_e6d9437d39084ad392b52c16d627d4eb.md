The CVE-2018-19499 report describes a PHP-specific bug class: an authenticated administrator triggering `unserialize()` on attacker-controlled data in Vanilla Forums' `Gdn_Format` class, leading to PHP object injection/RCE via magic methods (`__wakeup`, `__destruct`, etc.). This is fundamentally a PHP-object-injection issue tied to PHP's `unserialize()` gadget chains.

I searched protobuf-go for an analogous "reachable unsafe deserialization to arbitrary type instantiation" pattern — the closest conceptual match is `Any` message type resolution via `FindMessageByURL` in [1](#0-0)  and the equivalent logic in `encoding/protojson`'s `unmarshalAny` [2](#0-1)  and `encoding/prototext`'s `unmarshalExpandedAny` [3](#0-2) .

However, none of these constitute an analogous vulnerability under the stated threat model:
- Type resolution only selects among message types already registered/linked into the binary (a trusted schema), via `protoregistry.GlobalTypes` or a caller-supplied resolver — it does not instantiate arbitrary attacker-named classes or invoke arbitrary code paths like PHP's magic methods.
- The resulting decode is a normal protobuf unmarshal into a `proto.Message`, not an unserialize-style gadget chain that can trigger side-effecting method calls purely from data.
- The rules explicitly instruct rejecting cases involving trusted schema/resolver assumptions and require a genuine unsafe-memory or authz/confidentiality effect, none of which is present here.

There is no reachable, unprivileged code path in protobuf-go that mirrors PHP's `unserialize`-to-RCE broken invariant.

func UnmarshalNew(src *Any, opts proto.UnmarshalOptions) (dst proto.Message, err error) {
	if src.GetTypeUrl() == "" {
		return nil, protoimpl.X.NewError("invalid empty type URL")
	}
	if opts.Resolver == nil {
		opts.Resolver = protoregistry.GlobalTypes
	}
	r, ok := opts.Resolver.(protoregistry.MessageTypeResolver)
	if !ok {
		return nil, protoregistry.NotFound
	}
	mt, err := r.FindMessageByURL(src.GetTypeUrl())
	if err != nil {
		if err == protoregistry.NotFound {
			return nil, err
		}
		return nil, protoimpl.X.NewError("could not resolve %q: %v", src.GetTypeUrl(), err)
	}
	dst = mt.New().Interface()
	return dst, opts.Unmarshal(src.GetValue(), dst)
}

func (d decoder) unmarshalExpandedAny(typeURL string, pos int) ([]byte, error) {
	mt, err := d.opts.Resolver.FindMessageByURL(typeURL)
	if err != nil {
		return nil, d.newError(pos, "unable to resolve message [%v]: %v", typeURL, err)
	}
	// Create new message for the embedded message type and unmarshal the value
	// field into it.
	m := mt.New()
	if err := d.unmarshalMessage(m, true); err != nil {
		return nil, err
	}
	// Serialize the embedded message and return the resulting bytes.
	b, err := proto.MarshalOptions{
		AllowPartial:  true, // Never check required fields inside an Any.
		Deterministic: true,
	}.Marshal(m.Interface())
	if err != nil {
		return nil, d.newError(pos, "error in marshaling message into Any.value: %v", err)
	}
	return b, nil
}
