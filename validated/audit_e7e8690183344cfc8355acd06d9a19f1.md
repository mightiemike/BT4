    /// `timestamp_micros` is a Unix Epoch timestamp in microseconds.

            Ok(named_chain) => TimedFeaturesImpl::OnNamedChain {

                timestamp_micros,
            },
            Err(_) => TimedFeaturesImpl::EnableAll, // Unknown chain => enable all features by default.
        };
        Self {
            inner,
            override_: None,
        }
